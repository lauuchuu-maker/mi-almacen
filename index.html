function cerrarDia() { if (ventasHoy.length === 0) return mostrarAlerta("No hay ventas registradas el día de hoy."); mostrarConfirmacion("¿Seguro que deseas cerrar el día? Las ventas pasarán al historial.", function() { let totalDia = ventasHoy.reduce((a, b) => a + (b.total || 0), 0); let gananciaDia = ventasHoy.reduce((a, b) => a + (b.ganancia || 0), 0); let todoLoVendidoHoy = ventasHoy.map(v => v.descripcion).join(' | '); // Recolectamos todos los items vendidos en el día para guardarlos en el historial let itemsVendidosHoy = []; ventasHoy.forEach(v => { if (v.items) itemsVendidosHoy.push(...v.items); }); historial.push({ fecha: new Date().toLocaleDateString(), total: totalDia, ganancia: gananciaDia, descripcion: todoLoVendidoHoy, items: itemsVendidosHoy // Guardamos los items aquí }); db.ref('historial').set(historial); db.ref('ventasHoy').remove(); carrito = []; actualizarCarrito(); }); } function eliminarCierre(index) { mostrarConfirmacion("¿Eliminar este registro de cierre? Se devolverá el stock de los productos vendidos.", function() { let cierre = historial[index]; // Si el cierre tiene items guardados, devolvemos el stock if (cierre.items) { cierre.items.forEach(soldItem => { let prod = inventario.find(p => p.nombre === soldItem.nombre); if (prod) { prod.stock += soldItem.cant; } }); db.ref('almacen').set(inventario); } historial.splice(index, 1); db.ref('historial').set(historial); }); } ``` Con estos ajustes, cada vez que cierres el día, el sistema recordará exactamente qué se vendió. Así, cuando necesites borrar un cierre por error, el programa sabrá qué cantidades devolver al inventario de forma automática. ¡Espero que esto deje tu sistema funcionando perfectamente!
